function MovementModal({ snapshot, type, setType, selectedTool, setSelectedTool, selectedWorker, setSelectedWorker, selectedStorekeeper, setSelectedStorekeeper, quantity, setQuantity, note, setNote, destination, setDestination, condition, setCondition, message, onClose, onSubmit }: { snapshot: Snapshot; type: MovementType; setType: (value: MovementType) => void; selectedTool: number | ""; setSelectedTool: (value: number | "") => void; selectedWorker: number | ""; setSelectedWorker: (value: number | "") => void; selectedStorekeeper: number | ""; setSelectedStorekeeper: (value: number | "") => void; quantity: number; setQuantity: (value: number) => void; note: string; setNote: (value: string) => void; destination: string; setDestination: (value: string) => void; condition: string; setCondition: (value: string) => void; message: string; onClose: () => void; onSubmit: () => void }) {
  const tool = snapshot.tools.find((item) => item.id === selectedTool);
  return <div className="modal-backdrop" onMouseDown={(event) => { if (event.target === event.currentTarget) onClose(); }}><div className="modal-card ops-modal"><div className="modal-header"><div><div className="eyebrow"><span className="eyebrow-dot" />Trazabilidad del almacén</div><h2>{typeLabel(type)}</h2><p>Selecciona el trabajador y el almacenero responsable del movimiento.</p></div><button className="modal-close" onClick={onClose}><X size={18} /></button></div><div className="movement-tabs"><button type="button" className={type === "dispatch" ? "active" : ""} onClick={() => setType("dispatch")}><ArrowUpFromLine size={16} />Despacho</button><button type="button" className={type === "return" ? "active" : ""} onClick={() => setType("return")}><ArrowDownToLine size={16} />Recepción</button><button type="button" className={type === "transfer" ? "active" : ""} onClick={() => setType("transfer")}><RotateCcw size={16} />Traslado</button></div><div className="ops-form-grid"><SelectField label="Herramienta" name="tool" value={selectedTool} onChange={(event) => setSelectedTool(event.target.value ? Number(event.target.value) : "")} options={snapshot.tools.map((item) => ({ value: item.id, label: `${item.code} · ${item.name}` }))} /><SelectField label="Trabajador / código" name="worker" value={selectedWorker} onChange={(event) => setSelectedWorker(event.target.value ? Number(event.target.value) : "")} options={snapshot.workers.map((item) => ({ value: item.id, label: `${item.code} · ${item.fullName}` }))} /><SelectField label="Almacenero responsable" name="storekeeper" value={selectedStorekeeper} onChange={(event) => setSelectedStorekeeper(event.target.value ? Number(event.target.value) : "")} options={snapshot.storekeepers.map((item) => ({ value: item.id, label: `${item.code} · ${item.fullName}` }))} /><label className="ops-field"><span>Cantidad</span><div className="ops-number"><button type="button" onClick={() => setQuantity(Math.max(1, quantity - 1))}>−</button><input type="number" min="1" value={quantity} onChange={(event) => setQuantity(Math.max(1, Number(event.target.value)))} /><button type="button" onClick={() => setQuantity(quantity + 1)}>+</button></div></label><Field label="Destino / frente de trabajo" name="destination" placeholder="Opcional" value={destination} onChange={(event) => setDestination(event.target.value)} /><Field label="Condición / estado" name="condition" placeholder="Bueno, con desgaste..." value={condition} onChange={(event) => setCondition(event.target.value)} /><label className="ops-field ops-full"><span>Observaciones</span><textarea rows={2} value={note} onChange={(event) => setNote(event.target.value)} placeholder="Accesorios, daños, motivo..." /></label><div className="ops-movement-summary ops-full"><div className="ops-movement-summary-icon">{tool?.imageUrl ? <img src={tool.imageUrl} alt="" /> : <Hammer size={18} />}</div><div><strong>{tool?.name || "Selecciona una herramienta"}</strong><span>{tool ? `${tool.availableQuantity} de ${tool.quantity} disponibles · ${tool.code}` : "Sin selección"}</span></div></div><div className="ops-form-footer ops-full"><span className="ops-form-message">{message || (snapshot.storekeepers.length === 0 ? "Registra al menos un almacenero en Configuración inicial." : "")}</span><button type="button" className="button button-light" onClick={onClose}>Cancelar</button><button type="button" className="button button-primary" onClick={onSubmit}><Check size={15} />Confirmar {typeLabel(type).toLowerCase()}</button></div></div></div></div>;
}
